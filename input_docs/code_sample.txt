3. Core Implementation Files
​Backend: Java / Spring Boot 3 (TrackController.java)
​Provides audio streaming capabilities with standard byte-range request support:



package com.spotify.controller;



import com.spotify.model.Track;

import com.spotify.repository.TrackRepository;

import org.springframework.core.io.Resource;

import org.springframework.core.io.UrlResource;

import org.springframework.http.*;

import org.springframework.web.bind.annotation.*;



import java.nio.file.Path;

import java.nio.file.Paths;

import java.util.List;



@RestController

@RequestMapping("/api/v1/tracks")

@CrossOrigin(origins = "http://localhost:4200")

public class TrackController {



    private final TrackRepository trackRepository;

    private final Path storageLocation = Paths.get("media/audio");



    public TrackController(TrackRepository trackRepository) {

        this.trackRepository = trackRepository;

    }



    @GetMapping

    public ResponseEntity<List<Track>> getAllTracks() {

        return ResponseEntity.ok(trackRepository.findAll());

    }



    @GetMapping("/stream/{filename}")

    public ResponseEntity<Resource> streamAudio(@PathVariable String filename) {

        try {

            Path filePath = storageLocation.resolve(filename).normalize();

            Resource resource = new UrlResource(filePath.toUri());



            if (!resource.exists()) {

                return ResponseEntity.notFound().build();

            }



            return ResponseEntity.ok()

                    .contentType(MediaType.parseMediaType("audio/mpeg"))

                    .header(HttpHeaders.CONTENT_DISPOSITION, "inline; filename=\"" + resource.getFilename() + "\"")

                    .body(resource);

        } catch (Exception e) {

            return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR).build();

        }

    }

}

Frontend: Angular 15 Audio Service (audio-player.service.ts)

​Handles client-side audio controls and state management





:import { Injectable } from '@angular/core';

import { BehaviorSubject } from 'rxjs';



export interface Track {

  id: number;

  title: string;

  artist: string;

  audioUrl: string;

  coverArtUrl: string;

}



@Injectable({

  providedIn: 'root'

})

export class AudioPlayerService {

  private audio = new Audio();

  

  public currentTrack$ = new BehaviorSubject<Track | null>(null);

  public isPlaying$ = new BehaviorSubject<boolean>(false);

  public currentTime$ = new BehaviorSubject<number>(0);

  public duration$ = new BehaviorSubject<number>(0);



  constructor() {

    this.audio.addEventListener('timeupdate', () => {

      this.currentTime$.next(this.audio.currentTime);

    });



    this.audio.addEventListener('loadedmetadata', () => {

      this.duration$.next(this.audio.duration);

    });



    this.audio.addEventListener('ended', () => {

      this.isPlaying$.next(false);

    });

  }



  playTrack(track: Track): void {

    if (this.currentTrack$.value?.id !== track.id) {

      this.currentTrack$.next(track);

      this.audio.src = track.audioUrl;

      this.audio.load();

    }

    this.audio.play();

    this.isPlaying$.next(true);

  }



  pauseTrack(): void {

    this.audio.pause();

    this.isPlaying$.next(false);

  }



  seek(seconds: number): void {

    this.audio.currentTime = seconds;

  }

}







Frontend: Angular 15 Player Component (player.component.ts)



import { Component, OnInit } from '@angular/core';

import { AudioPlayerService, Track } from '../../core/services/audio-player.service';



@Component({

  selector: 'app-player',

  templateUrl: './player.component.html',

  styleUrls: ['./player.component.scss']

})

export class PlayerComponent implements OnInit {

  currentTrack: Track | null = null;

  isPlaying = false;

  currentTime = 0;

  duration = 0;



  constructor(public playerService: AudioPlayerService) {}



  ngOnInit(): void {

    this.playerService.currentTrack$.subscribe(track => this.currentTrack = track);

    this.playerService.isPlaying$.subscribe(status => this.isPlaying = status);

    this.playerService.currentTime$.subscribe(time => this.currentTime = time);

    this.playerService.duration$.subscribe(dur => this.duration = dur);

  }



  togglePlayPause(): void {

    if (this.isPlaying) {

      this.playerService.pauseTrack();

    } else if (this.currentTrack) {

      this.playerService.playTrack(this.currentTrack);

    }

  }



  onSeek(event: Event): void {

    const input = event.target as HTMLInputElement;

    this.playerService.seek(Number(input.value));

  }

}
